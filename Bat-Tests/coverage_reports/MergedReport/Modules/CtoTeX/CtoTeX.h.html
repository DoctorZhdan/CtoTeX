/// @brief Ñòðóêòóðà äëÿ õðàíåíèÿ äàííûõ îá îøèáêå
struct Error {
    ErrorCode code = InputFileNotExist; ///< Êîä îøèáêè èç ïåðå÷èñëåíèÿ
    int position = -1;       ///< Èíäåêñ ñèìâîëà, ñîäåðæàùåãî îøèáêó, â ñòðîêå 
    int length = 1;          ///< Äëèíà ïîñëåäîâàòåëüíîñòè íåêîððåêòíûõ ñèìâîëîâ 
    string line;             ///< Ñòðîêà, ñîäåðæàùàÿ îøèáêó

    /// @brief Ìåòîä ãåíåðàöèè ñîîáùåíèÿ îá îøèáêå
    /// @return ñôîðìèðîâàííîå ñîîáùåíèå îá îøèáêå
    string generateErrorMessage() const {
        // 1. Âçÿòü áàçîâîå ñîîáùåíèå îá îøèáêå, ñîîòâåòñòâóþùåå êîäó îøèáêè
        string result = errorMessages.at(code);

        // 2. Åñëè èçâåñòíà ïîçèöèÿ îøèáêè â ñòðîêå è ñàìà ñòðîêà íå ïóñòàÿ
        if (position != -1 && !line.empty()) {
            // 2.1. Äîáàâèòü ê áàçîâîìó ñîîáùåíèþ ñèìâîë ïåðåõîäà íà íîâóþ ñòðîêó è èñõîäíóþ ñòðîêó
            result += "\n" + line;

            // 2.2. Ñîçäàòü ïóñòóþ ñòðîêó-óêàçàòåëü
            string pointer;

            // 2.3. Äîáàâèòü â ñòðîêó-óêàçàòåëü ñòîëüêî ñèìâîëîâ '-', ñêîëüêî ïîçèöèé íóæíî ïðîïóñòèòü äî ìåñòà îøèáêè
            for (int i = 0; i < position; ++i) {
                pointer += '-';
            }

            // 2.4. Äîáàâèòü â ñòðîêó-óêàçàòåëü ñòîëüêî ñèìâîëîâ '^', êàêîâà äëèíà ïîñëåäîâàòåëüíîñòè íåêîððåêòíûõ ñèìâîëîâ
            for (int i = 0; i < length; ++i) {
                pointer += '^';
            }

            // 2.5. Äîáàâèòü ê ñîîáùåíèþ ñèìâîë ïåðåõîäà íà íîâóþ ñòðîêó è ñòðîêó-óêàçàòåëü
            result += "\n" + pointer;
        }

        // 3. Âåðíóòü ïîëó÷åííîå ñîîáùåíèå
        return result;
    }
};

/// @brief Ñòðóêòóðà òîêåíà (ëåêñåìû)
struct Token {
    TokenType type = UNKNOWN; ///< òèï òîêåíà
    string value;            ///< òåêñòîâîå ïðåäñòàâëåíèå òîêåíà
    OperandType operandType = ARITHMETIC; ///< òèï îïåðàíäîâ (àðèôìåòè÷åñêèé èëè ëîãè÷åñêèé)
};

/// @brief Êëàññ óçëà äåðåâà ðàçáîðà
class Node {
public:
    Token token;             ///< òîêåí óçëà
    Node* left = nullptr;    ///< óêàçàòåëü íà ëåâûé ïîòîìîê (åäèíñòâåííûé äëÿ óíàðíûõ îïåðàöèé) 
    Node* right = nullptr;   ///< óêàçàòåëü íà ïðàâûé ïîòîìîê (nullptr äëÿ óíàðíûõ îïåðàöèé)

    /// @brief Êîíñòðóêòîð äëÿ ëèñòà (÷èñëî, ïåðåìåííàÿ, êîíñòàíòà)
    /// @param t òîêåí
    Node(const Token& t) : token(t) {}

    /// @brief Êîíñòðóêòîð äëÿ áèíàðíîé îïåðàöèè
    /// @param t òîêåí îïåðàöèè
    /// @param l ëåâûé îïåðàíä
    /// @param r ïðàâûé îïåðàíä
    Node(const Token& t, Node* l, Node* r) : token(t), left(l), right(r) {}

    /// @brief Êîíñòðóêòîð äëÿ óíàðíîé îïåðàöèè
    /// @param t òîêåí îïåðàöèè
    /// @param l åäèíñòâåííûé îïåðàíä
    Node(const Token& t, Node* l) : token(t), left(l) {}

    /// @brief Äåñòðóêòîð, ðåêóðñèâíî óäàëÿþùèé äî÷åðíèå óçëû
    ~Node() {
        delete left;
        delete right;
    }

    /// @brief Çàïðåò êîíñòðóêòîðà êîïèðîâàíèÿ
    Node(const Node&) = delete;

    /// @brief Çàïðåò îïåðàòîðà ïðèñâàèâàíèÿ
    Node& operator=(const Node&) = delete;

};
