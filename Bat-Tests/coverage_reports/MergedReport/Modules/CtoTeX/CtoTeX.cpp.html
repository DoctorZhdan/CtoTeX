bool checkOperatorSpacing(const string& word, vector<Error>& errors, const set<string>& allowedOperations)
{
    // 1. ÐÐ¾Ð´ÑÑÑÑ ÐºÐ¾Ð»Ð¸ÑÐµÑÑÐ²Ð° Ð¾Ð¿ÐµÑÐ°ÑÐ¾ÑÐ¾Ð² Ð² ÑÐ»Ð¾Ð²Ðµ 
    int opCounter = 0;

    for (const string& op : allowedOperations) {
        size_t pos = 0;
        while ((pos = word.find(op, pos)) != string::npos) {
            opCounter++;
            pos += op.size();
        }
    }

    // 2. ÐÑÐ¾Ð²ÐµÑÐºÐ° Ð½Ð° Ð½Ð°Ð»Ð¸ÑÐ¸Ðµ Ð¾Ð¿ÐµÑÐ°ÑÐ¾ÑÐ° Ð²Ð½ÑÑÑÐ¸ ÑÐ»Ð¾Ð²Ð° 
    if (opCounter > 1 || (opCounter == 1 && allowedOperations.find(word) == allowedOperations.end()))
    {
        //  3. ÐÑÐ¾Ð²ÐµÑÐºÐ°, Ð½Ðµ ÑÐ²Ð»ÑÐµÑÑÑ Ð»Ð¸ ÑÐ»Ð¾Ð²Ð¾ Ð¾ÑÑÐ¸ÑÐ°ÑÐµÐ»ÑÐ½ÑÐ¼ ÑÐ¸ÑÐ»Ð¾Ð¼ 
        bool isNegativeNumber = false;

        if (word.size() > 1 && word[0] == '-') {
            isNegativeNumber = true;
            bool dotFound = false;

            for (size_t i = 1; i < word.size(); i++) {
                if (isdigit(word[i])) {}
                else if (word[i] == '.' && !dotFound) {
                    dotFound = true;
                }
                else {
                    isNegativeNumber = false;
                }
            }
        }

        //  4. ÐÑÐ»Ð¸ ÑÑÐ¾ Ð½Ðµ ÑÐ¸ÑÐ»Ð¾, Ð¾ÑÐ¸Ð±ÐºÐ°
        if (!isNegativeNumber)
        {
            Error err;
            err.code = (opCounter > 1) ? InvalidSymbolSequence : OperatorNotSeparatedBySpaces;
            err.position = 0;
            err.length = word.size();
            err.line = word;

            errors.push_back(err);
            return true;
        }
    }

    return false;
}

    // 2. ÐÑÐ¾Ð²ÐµÑÐºÐ° ÑÑÑÑÐºÑÑÑÑ ÑÐ¸ÑÐ»Ð°

    }

bool parseArray(const string& word, vector<Token>& tokens, vector<Error>& errors, int& nodeCount)

    size_t l = word.find('[');

    Token arrayToken;

Node* processBinaryOperator(const Token& token, stack<Node*>& nodeStack, vector<Error>& errors)

    if (nodeStack.size() < 2)

        return nullptr;
    }

    // 2. ÐÐ·Ð²Ð»ÐµÑÐµÐ½Ð¸Ðµ Ð¾Ð¿ÐµÑÐ°Ð½Ð´Ð¾Ð²
    Node* right = nodeStack.top();
    nodeStack.pop();

    }

    return new Node(token, left, right);
