90 string generateErrorMessage() const {
91 // 1. Взять базовое сообщение об ошибке, соответствующее коду ошибки
92 string result = errorMessages.at(code);
93
94 // 2. Если известна позиция ошибки в строке и сама строка не пустая
95 if (position != -1 && !line.empty()) {
96 // 2.1. Добавить к базовому сообщению символ перехода на новую строку и исходную строку
97 result += "\n" + line;
98
99 // 2.2. Создать пустую строку-указатель
100 string pointer;
101
102 // 2.3. Добавить в строку-указатель столько символов '-', сколько позиций нужно пропустить до места ошибки
103 for (int i = 0; i < position; ++i) {
104 pointer += '-';
105 }
106
107 // 2.4. Добавить в строку-указатель столько символов '^', какова длина последовательности некорректных символов
108 for (int i = 0; i < length; ++i) {
109 pointer += '^';
110 }
111
112 // 2.5. Добавить к сообщению символ перехода на новую строку и строку-указатель
113 result += "\n" + pointer;
114 }
115
116 // 3. Вернуть полученное сообщение
117 return result;
118 }
